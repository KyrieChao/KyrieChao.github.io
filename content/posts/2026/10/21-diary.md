    *returnSize = num;
    for (int i = 0; i < num; i++)

```c
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* getRow(int rowIndex, int* returnSize) {
    int num = rowIndex + 1;
    int* arr = malloc(sizeof(int) * num);
    arr[0] = arr[num - 1] = 1;
    *returnSize = num;
    if (num < 2) {
        return arr;
    }
    int n1;
    int* brr = getRow(num - 2, &n1); // 这里 num -1 会死循环  
    // 这里我想拿到的是arr的上一层 num是rowIndex + 1 
    // num -1 一直求rowIndex的
    // 没有求到rowIndex-1的 所以 疏忽了
    for (int i = 1; i < num - 1; i++) {
        arr[i] = brr[i - 1] + brr[i];
    }
    free(brr);
    return arr;
}
```

又解决一道 虽然很简单 但是苍蝇再小也是肉
```c
// 给定一个非负整数 num，反复将各个位上的数字相加，直到结果为一位数。返回这个结果。

// 示例 :
// 输入: num = 38
// 输出: 2 
// 解释: 各位相加的过程为：
// 38 --> 3 + 8 --> 11
// 11 --> 1 + 1 --> 2
// 由于 2 是一位数，所以返回 2。

int addDigits(int num) {
    int n = 0;
    while (1) {
        n += num % 10;
        num /= 10;
        if (num <= 0)
            break;
    }
    if (n >= 10) {
        return addDigits(n);
    }else{
        return n;
    }

}
```
